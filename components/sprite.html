Sprite Component

render sprite of an entity.

Color

change every float percante between 0-255 to make the Color

Name Type Description
r float Red percante
g float Green percante
b float Blue percante
a float Alpha percante

Example


-- Create new Color. red, green, blue, alpha.
local color = Color(255, 255, 255, 255)--White

Properties

Name Type Description
texture string name texture or image.png/.jpg to sprite renderer
layer int high layer up on low layer
color Color sprite color from rgba
flipX bool flip dir horizontal sprite render
flipY bool flip dir vertical sprite render

Example


-- Create a new entity
local entity = Entity("name", "group")

-- Add a new sprite component at texture name, layer, hidden, color. 
local sprite = entity:AddComponent(Sprite("none", 0, false, Color(255, 255, 255, 255)--white))

-- access the sprite properties
sprite.texture = "idle.png"
sprite.layer = 2
sprite.color = Color(255, 0, 0, 255)--red
sprite.flipX = true

Sprite Component

لعرض الصور في العالم

Color

كل متغير لون واحد من float يمكن تغيره ما بين 0-255 الى تكون لون واحد محدد

اسم المتغير النوع الوصف
r float نسبة من الون الاحمر
g float نسبة من الون الاخضر
b float نسبة من الون الازرق
a float نسبة مدى الشفافية

مثال


-- انشاء لون جديد. red, green, blue, alpha.
local color = Color(255, 255, 255, 255)--ابيض

المتغيرات

اسم المتغير النوع الوصف
texture string اسم الصورة للاستدعا texture وعرضة في العالم
layer int الطبقة الرقم يحدد اولوية العرض
color Color لون الصورة
flipX bool لعكس الصورة بشكل افقي
flipY bool لغكس الصورة بشكل عامودي

مثال


-- انشاء كائن جديد
local entity = Entity("name", "group")

-- اضافة مكون Sprite, texture, layer, hidden, color
local sprite = entity:AddComponent(Sprite("none", 0, false, Color(255, 255, 255, 255)--ابيض))

-- access the sprite properties
sprite.texture = "idle.png"
sprite.layer = 2
sprite.color = Color(255, 0, 0, 255)--احمر
sprite.flipX = true
